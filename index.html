/** * Scanner Barcode Invoice - API (Google Apps Script) * Kolom: A Timestamp | B No Invoice | C Nilai Invoice | D User | E Status */ const SHEET_NAME = 'Invoice'; const HEADERS = ['Timestamp', 'No Invoice', 'Nilai Invoice', 'User', 'Status']; // Cache per eksekusi: dibuka sekali saja, dipakai ulang let _props, _sheet; function doGet() { return json_({ ok: true, message: 'API Scanner Invoice aktif.' }); } function doPost(e) { let req; try { req = JSON.parse(e.postData.contents); } catch (err) { return json_({ ok: false, message: 'Permintaan tidak valid.' }); } const pin = props_().getProperty('API_PIN'); if (!pin || String(req.pin) !== pin) { return json_({ ok: false, code: 'PIN', message: 'PIN salah.' }); } if (req.action === 'check') return json_(checkInvoice(req.invoice)); if (req.action === 'save') return json_(saveInvoice(req)); return json_({ ok: false, message: 'Aksi tidak dikenal.' }); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function props_() { return _props || (_props = PropertiesService.getScriptProperties()); } /** Jalankan sekali. Ganti PIN sesuai keinginan. */ function setPin() { props_().setProperty('API_PIN', '1705'); } /** Opsional: jalankan sekali jika script TIDAK terikat ke spreadsheet. */ function setSpreadsheetId() { props_().setProperty('SPREADSHEET_ID', 'GANTI_DENGAN_ID_SPREADSHEET'); } /** Sheet dibuka sekali per eksekusi; dibuat otomatis beserta header jika belum ada. */ function getSheet_() { if (_sheet) return _sheet; const id = props_().getProperty('SPREADSHEET_ID'); const ss = id ? SpreadsheetApp.openById(id) : SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold'); sheet.setFrozenRows(1); sheet.getRange('B:B').setNumberFormat('@'); // seluruh kolom B = teks (sekali, saat dibuat) } return (_sheet = sheet); } /** Cari baris berdasarkan No Invoice dengan TextFinder (tanpa membaca seluruh kolom). */ function findRow_(sheet, invoice) { const last = sheet.getLastRow(); if (last < 2) return 0; const hit = sheet.getRange(2, 2, last - 1, 1) .createTextFinder(invoice) .matchEntireCell(true) .findNext(); return hit ? hit.getRow() : 0; } function checkInvoice(invoice) { invoice = String(invoice || '').trim(); if (!invoice) return { ok: true, exists: false }; const sheet = getSheet_(); const row = findRow_(sheet, invoice); if (!row) return { ok: true, exists: false }; return { ok: true, exists: true, nilai: sheet.getRange(row, 3).getValue() }; } function saveInvoice(data) { const invoice = String((data && data.invoice) || '').trim(); const harga = Number(data && data.harga); const user = String((data && data.user) || '').trim() || '-'; if (!invoice) return { ok: false, message: 'Nomor invoice kosong.' }; if (!isFinite(harga) || harga <= 0) return { ok: false, message: 'Nilai invoice harus lebih dari 0.' }; const lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (e) { return { ok: false, message: 'Server sedang sibuk. Coba lagi.' }; } try { const sheet = getSheet_(); const row = findRow_(sheet, invoice); if (row) { // 1 panggilan tulis untuk seluruh baris (sebelumnya 4) sheet.getRange(row, 1, 1, 5).setValues([[new Date(), invoice, harga, user, 'Dikoreksi']]); return { ok: true, message: 'Data diperbarui.' }; } // 1 panggilan tulis; tidak perlu getLastRow + setNumberFormat per baris sheet.appendRow([new Date(), invoice, harga, user, 'Tersimpan']); return { ok: true, message: 'Data tersimpan.' }; } catch (err) { return { ok: false, message: 'Gagal menyimpan: ' + err.message }; } finally { lock.releaseLock(); } }
